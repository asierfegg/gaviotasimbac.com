# 1. Eliminar servicios del registro
$servicios = @("VMTools","VGAuthService","VM3DService","vmvss","VMwareCAF","hgfs")
foreach ($s in $servicios) {
    sc.exe delete $s 2>$null
    Write-Host "Eliminado servicio: $s"
}

# 2. Eliminar carpetas de instalación
$rutas = @(
    "C:\Program Files\VMware",
    "C:\Program Files (x86)\VMware",
    "C:\ProgramData\VMware"
)
foreach ($r in $rutas) {
    if (Test-Path $r) {
        Remove-Item $r -Recurse -Force
        Write-Host "Eliminado: $r"
    }
}


# 3. Limpiar el registro MSI (esto es lo clave para que Windows no lo vea como instalado)
$guid = "{298083DA-1DC8-4886-97DD-333954302E68}"
$guidInverso = "AD380892-8CD1-6884-79DD-333954302E68"  # GUID invertido para Installer

Remove-Item "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall\$guid" -Force -ErrorAction SilentlyContinue
Remove-Item "HKLM:\SOFTWARE\Classes\Installer\Products\$guidInverso" -Force -ErrorAction SilentlyContinue
Remove-Item "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Installer\UserData\S-1-5-18\Products\$guidInverso" -Recurse -Force -ErrorAction SilentlyContinue


# 4. Eliminar drivers VMware del sistema
$drivers = @("vmxnet","vmxnet3","vmci","vsock","vmmouse","vmscsi","pvscsi")
foreach ($d in $drivers) {
    pnputil /delete-driver $d.inf /uninstall /force 2>$null
}
